cmake_minimum_required(VERSION 3.17)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(runner_state_tests LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(RNBO_DIR "" CACHE PATH "RNBO source directory containing RNBO.h")
    enable_testing()
endif()
if(NOT EXISTS "${RNBO_DIR}/RNBO.h")
    message(FATAL_ERROR "Set RNBO_DIR to the runner's RNBO source package")
endif()
add_executable(set_dirty_state_test SetDirtyStateTest.cpp)
target_include_directories(set_dirty_state_test PRIVATE ../src "${RNBO_DIR}" "${RNBO_DIR}/common" "${RNBO_DIR}/src" "${RNBO_DIR}/src/3rdparty")
add_test(NAME set_dirty_state COMMAND set_dirty_state_test)
add_executable(midi_connections_test MidiConnectionsTest.cpp)
target_include_directories(midi_connections_test PRIVATE ../src "${RNBO_DIR}" "${RNBO_DIR}/common" "${RNBO_DIR}/src" "${RNBO_DIR}/src/3rdparty")
add_test(NAME midi_connections COMMAND midi_connections_test)

find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME midi_migration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/MidiMigrationTest.py")
endif()

option(RUNNER_JACK_TESTS "Build integration test requiring an isolated running JACK server" OFF)
if(RUNNER_JACK_TESTS)
    find_package(Threads REQUIRED)
    find_path(MIDI_TEST_JACK_INCLUDE jack/jack.h REQUIRED)
    find_library(MIDI_TEST_JACK_LIBRARY jack REQUIRED)
    add_executable(jack_midi_connections_test JackMidiConnectionsTest.cpp)
    target_include_directories(jack_midi_connections_test PRIVATE ../src "${MIDI_TEST_JACK_INCLUDE}" "${RNBO_DIR}" "${RNBO_DIR}/common" "${RNBO_DIR}/src" "${RNBO_DIR}/src/3rdparty")
    target_link_libraries(jack_midi_connections_test PRIVATE "${MIDI_TEST_JACK_LIBRARY}" Threads::Threads)
    add_test(NAME jack_midi_connections COMMAND jack_midi_connections_test)
endif()

# RNBO.h defines preset utilities that these standalone tests do not use. On
# ELF platforms discard those sections instead of linking the DSP engine.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(state_test set_dirty_state_test midi_connections_test jack_midi_connections_test)
        if(TARGET ${state_test})
            target_compile_options(${state_test} PRIVATE -ffunction-sections -fdata-sections)
            target_link_options(${state_test} PRIVATE -Wl,--gc-sections)
        endif()
    endforeach()
endif()
